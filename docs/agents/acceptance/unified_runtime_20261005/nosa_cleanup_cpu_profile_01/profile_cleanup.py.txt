def summary(values):
    median=statistics.median(values)
    return {'values':values,'median':median,'min':min(values),'max':max(values),'range':max(values)-min(values)}
def timed(fn):
    for _ in range(20):fn()
    samples=[]
    for _ in range(7):
        begin=time.perf_counter_ns()
        for _ in range(500):fn()
        samples.append((time.perf_counter_ns()-begin)/500/1000)
    return summary(samples)
def profile(fn,name):
    p=cProfile.Profile();p.enable()
    for _ in range(300):fn()
    p.disable();p.dump_stats(out/f'{role}_{name}.pstats')
    rows=[]
    for (file,line,function),(primitive,total,self_seconds,cumulative,callers) in pstats.Stats(p).stats.items():
        rows.append({'file':file,'line':line,'function':function,'calls_per_iteration':total/300,
                     'self_us_per_iteration':self_seconds*1e6/300,'cumulative_us_per_iteration':cumulative*1e6/300})
    return sorted(rows,key=lambda r:r['cumulative_us_per_iteration'],reverse=True)

cases={}
for count in (1,16):
    backend=SharedBackend()
    original_plan=backend.plan_resources
    def planning(budgets,limits):return replace(original_plan(budgets,limits),host_pages=count)
    backend.plan_resources=planning
    output=torch.tensor([[6],[10]])
    def no_compute(session,ids):
        session['tokens'].extend((3,4))
        return output
    backend.extend=no_compute
    with PersistentGRRunner(backend,resource_limits={'max_session_capacity':4}) as runner:
        for user in range(count):runner.execute(request(user))
        assert len(runner.pool)==count
        req=request(0)
        def execute():return runner.execute(req)
        audit=timed(runner.pool.audit)
        for _ in range(20):execute()
        cleanup=[];candidate=[];wall=[]
        for _ in range(7):
            results=[execute().metrics for _ in range(500)]
            cleanup.append(statistics.fmean(r['cleanup_ms'] for r in results)*1000)
            candidate.append(statistics.fmean(r['extend_ms'] for r in results)*1000)
            wall.append(statistics.fmean(r['latency_ms'] for r in results)*1000)
        cases[str(count)]={'pool_audit_us':audit,'mock_candidate_cleanup_us':summary(cleanup),
            'mock_candidate_extend_us':summary(candidate),'mock_request_wall_us':summary(wall),
            'profile_pool_audit':profile(runner.pool.audit,f'{count}_audit'),
            'profile_mock_execute':profile(execute,f'{count}_execute')}
    backend.close()
assert not torch.cuda.is_initialized()

    'unit':'microseconds','command':sys.orig_argv,'affinity':sorted(os.sched_getaffinity(0)),
    'threads':torch.get_num_threads(),'gc_enabled':gc.isenabled(),'cuda_initialized':torch.cuda.is_initialized(),
