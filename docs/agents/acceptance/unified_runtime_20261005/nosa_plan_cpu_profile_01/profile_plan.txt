identity = ('cpu-profile-history',65536)

def measure(fn, repeats=7, iterations=100):
    for _ in range(10): fn()
    samples=[]
    for _ in range(repeats):
        start=time.perf_counter_ns()
        for _ in range(iterations): fn()
        samples.append((time.perf_counter_ns()-start)/iterations/1000)
    return {'unit':'microseconds_per_call','batch_means':samples,
            'median':statistics.median(samples),'min':min(samples),'max':max(samples),
            'repeats':repeats,'iterations_per_repeat':iterations}

cases={}
for scheme in session_budget.SCHEMES:
    resources=OfflineNosaFixedResources(scheme=scheme,layers=32,
        kv_heads=config.num_key_value_heads,query_heads=config.num_attention_heads,
        head_dim=config.head_dim,max_seq_len=65664,chunk_size=1024,
        sparse_pool_tokens=65536,host_arena_tokens=16777216,
        device='cuda:0',dtype=torch.bfloat16)
    resource_plan=resources.plan_resources(None,limits_dict)
    resource_plan=replace(resource_plan, limits=ExecutionLimits.from_mapping(
        limits_dict,context=65664,chunk_size=1024))
    backend=object.__new__(NosaFixedServingBackend)
    backend.config=config;backend.resources=resources;backend.scheme=scheme
    backend.device=torch.device('cuda:0');backend.dtype=torch.bfloat16
    backend.chunk_size=1024;backend.max_seq_len=65664
    def plan(): return backend.plan_runtime_session(resource_plan,shape,identity)
    options=dict(capacity=65664,queries=1024,layers=32,
        kv_heads=config.num_key_value_heads,query_heads=config.num_attention_heads,
        head_dim=config.head_dim,dtype=torch.bfloat16,device='cuda:0',scheme=scheme,
        host_capacity=65536,dense_counter=True)
    def layout(): return session_budget.session_allocation_layout(**options)
    established=plan();allocations,breakdown=layout()
    assert established.allocations==allocations
    assert (established.reservation.hbm,established.reservation.dram)==(breakdown['hbm'],breakdown['dram'])
    spec_values=[{f.name:getattr(item,f.name) for f in fields(AllocationSpec)} for item in allocations]
    plan_values={f.name:getattr(established,f.name) for f in fields(SessionPlan)}
    original=session_budget.allocation_footprint;internal=[]
    def record_scan(items):
        internal.append(items)
        return original(items)
    session_budget.allocation_footprint=record_scan
    try: layout()
    finally: session_budget.allocation_footprint=original
    funcs={
        'plan_runtime_session':plan,
        'session_allocation_layout':layout,
        'all_internal_layout_footprint_scans':lambda:[allocation_footprint(items) for items in internal],
        'one_full_allocation_footprint':lambda:allocation_footprint(allocations),
        'construct_all_AllocationSpec_from_prebuilt_kwargs':lambda:tuple(AllocationSpec(**values) for values in spec_values),
        'validate_all_existing_AllocationSpec':lambda:[AllocationSpec.__post_init__(item) for item in allocations],
        'construct_SessionPlan_from_prebuilt_kwargs':lambda:SessionPlan(**plan_values),
        'validate_existing_SessionPlan':lambda:SessionPlan.__post_init__(established),
    }
    timing={name:measure(fn) for name,fn in funcs.items()}
    profiler=cProfile.Profile();profiler.enable()
    for _ in range(200): plan()
    profiler.disable();profiler.dump_stats(OUT/f'{scheme}.pstats')
    stats=pstats.Stats(profiler)
    profile_rows=[]
    for (filename,line,function),(primitive,total,self_time,cumulative,callers) in stats.stats.items():
        profile_rows.append({'file':filename,'line':line,'function':function,
            'primitive_calls_per_plan':primitive/200,'total_calls_per_plan':total/200,
            'self_us_per_plan':self_time*1e6/200,'cumulative_us_per_plan':cumulative*1e6/200,
            'callers':[{'file':key[0],'line':key[1],'function':key[2],'stats':value} for key,value in callers.items()]})
    cases[scheme]={'allocation_count':len(allocations),
        'internal_footprint_scan_lengths':[len(items) for items in internal],
        'external_full_footprint_scans_per_plan':2,
        'reservation_bytes':{'hbm':established.reservation.hbm,'dram':established.reservation.dram},
        'timing':timing,'profile_sorted_cumulative':sorted(profile_rows,key=lambda r:r['cumulative_us_per_plan'],reverse=True)}
    print(json.dumps({'scheme':scheme,'allocation_count':len(allocations),
        'timing_median_us':{k:v['median'] for k,v in timing.items()}}),flush=True)
